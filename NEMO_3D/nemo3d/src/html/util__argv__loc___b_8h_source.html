269/* declare the arguments needed to maintain a queue for type TYPE */
270#define QUEUE_DECLARE(type) \
271 type *queue_alloc = NULL; \
272 int queue_head = 0; \
273 int queue_tail = 0; \
274 int queue_count = 0; \
275 int queue_max = 1
276#define LOCAL_QUEUE_DECLARE(type) \
277 static type *queue_alloc = NULL; \
278 static int queue_head = 0; \
279 static int queue_tail = 0; \
280 static int queue_count = 0; \
281 static int queue_max = 1
282
283/* test to see if the queue has not been allocated */
284#define QUEUE_IS_NULL() (queue_alloc == NULL)
285
286/* allocate a mini-queue */
287#define QUEUE_ALLOC(type, num) \
288 do { \
289 queue_alloc = (type *)malloc(sizeof(type) * num); \
290 queue_head = 0; \
291 queue_tail = 0; \
292 queue_count = 0; \
293 queue_max = num; \
294 } while(0)
295
296/* allocate a mini-queue's allocation */
297#define QUEUE_FREE() free(queue_alloc)
298
299/* enqueue items from the mini-queue */
300#define QUEUE_ENQUEUE(item) \
301 do { \
302 queue_alloc[queue_head] = item; \
303 queue_head = (queue_head + 1) % queue_max; \
304 queue_count++; \
305 } while(0)
306
307/* dequeue items from the mini-queue */
308#define QUEUE_DEQUEUE(item) \
309 do { \
310 item = queue_alloc[queue_tail]; \
311 queue_tail = (queue_tail + 1) % queue_max; \
312 queue_count--; \
313 } while(0)
314
315/* count the number of items in the mini-queue */
316#define QUEUE_COUNT() queue_count
