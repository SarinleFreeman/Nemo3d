684 real phaseBloch(real kxL, real kyL, real kzL, int l, int m, int n) {
685 real phase = 0.0;
686 if (!is_tilted) { /* Calculate Bloch phase as follows when lattice is NOT tilted */
687 const int* dCell = NbrCell(m,n);
688 if ( isInX_min(l,m) && dCell[0]==-1 ) phase -= kxL;
689 if ( isInX_max(l,m) && dCell[0]== 1 ) phase += kxL;
690 if ( isInY_min(l,m) && dCell[1]==-1 ) phase -= kyL;
691 if ( isInY_max(l,m) && dCell[1]== 1 ) phase += kyL;
692 if ( isInZ_min(l,m) && dCell[2]==-1 ) phase -= kzL;
693 if ( isInZ_max(l,m) && dCell[2]== 1 ) phase += kzL;
694 } /* if !is_tilted */
695
696 if (is_tilted) { /* Calculate Bloch phase as follows when lattice is tilted */
697 /* Initialize all booleans to false */
698 bool _isNotInXnegtv = false;
699 bool _isNotInXpostv = false;
700 bool _isNotInYnegtv = false;
701 bool _isNotInYpostv = false;
702 bool _isNotInZnegtv = false;
703 bool _isNotInZpostv = false;
704
705 isNbrOut(l, m, n, &_isNotInXnegtv, &_isNotInXpostv, &_isNotInYnegtv,
706 &_isNotInYpostv, &_isNotInZnegtv, &_isNotInZpostv);
707
708 if (_isNotInXnegtv) {
709 phase -= kxL;
710// if(isPeriodicY()) phase -= ((double)dy/(double)dx)*kxL;
711 if(isPeriodicZ()) phase -= (((double)dz*lattice_z)/((double)dx*lattice_x))*kzL; /* Modified for ZB_111: scaled by lattice constants */
712 }
713 if (_isNotInXpostv) {
714 phase += kxL;
715// if(isPeriodicY()) phase += ((double)dy/(double)dx)*kxL;
716 if(isPeriodicZ()) phase += (((double)dz*lattice_z)/((double)dx*lattice_x))*kzL; /* Modified for ZB_111: scaled by lattice constants */
717 }
718 if (_isNotInYnegtv) { phase -= kyL; }
719 if (_isNotInYpostv) { phase += kyL; }
720 if (_isNotInZnegtv) { phase -= kzL; }
721 if (_isNotInZpostv) { phase += kzL; }
722 } /* if is_tilted */
723
724 return -Pi * (phase - 2*int(0.5*phase));
725 }

741 real phasePeriodic(real kxL, real kyL, real kzL,
742 int i_nbr, int j_nbr, int k_nbr) {
743 real phase = 0.0;
744 if (i_nbr < this->cell_xmin) phase -= kxL;
745 if (i_nbr > this->cell_xmax) phase += kxL;
746 if (j_nbr < this->cell_ymin) phase -= kyL;
747 if (j_nbr > this->cell_ymax) phase += kyL;
748 if (k_nbr < this->cell_zmin) phase -= kzL;
749 if (k_nbr > this->cell_zmax) phase += kzL;
750 return -Pi * (phase - 2*int(0.5*phase));
751 }
